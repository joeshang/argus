cmake_minimum_required(VERSION 3.24)

# Set before project(), and before fetching dependencies, so every object uses
# the same deployment target and requested CMAKE_OSX_ARCHITECTURES.
if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.3" CACHE STRING "Minimum macOS version")
endif()
project(Argus VERSION 2.1.0 LANGUAGES CXX OBJC OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The native Argus implementation currently requires macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

set(ARGUS_SANITIZER "none" CACHE STRING "none, undefined, address, thread, address,undefined")
set_property(CACHE ARGUS_SANITIZER PROPERTY STRINGS none undefined address thread "address,undefined")
if(NOT ARGUS_SANITIZER MATCHES "^(none|undefined|address|thread|address,undefined)$")
  message(FATAL_ERROR "Unsupported ARGUS_SANITIZER: ${ARGUS_SANITIZER}")
endif()
if(NOT ARGUS_SANITIZER STREQUAL "none")
  # Directory-wide flags also instrument FTXUI, which is now built from source.
  add_compile_options("-fsanitize=${ARGUS_SANITIZER}" -fno-omit-frame-pointer)
  add_link_options("-fsanitize=${ARGUS_SANITIZER}" -fno-omit-frame-pointer)
endif()
# Ship self-contained executables; fetched FTXUI libraries stay static.
set(BUILD_SHARED_LIBS OFF)
include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)
find_library(ARGUS_FOUNDATION Foundation REQUIRED)
find_library(ARGUS_CORE_SERVICES CoreServices REQUIRED)

add_library(argus_warnings INTERFACE)
target_compile_options(argus_warnings INTERFACE -Wall -Wextra -Werror -Wno-unused-parameter -Wno-sign-compare)
set(service "${CMAKE_CURRENT_SOURCE_DIR}/argus-native-service/ArgusMacServices/ArgusMacServices")
set(helper "${CMAKE_CURRENT_SOURCE_DIR}/argus-extension-helper/ArgusMacExtensionHelper/ArgusMacExtensionHelper")

add_library(argus_monitor STATIC "${service}/BrowserMonitor.cpp" "${service}/ProcessUtils.mm" "${service}/WebKitUtils.m")
target_include_directories(argus_monitor PUBLIC "${service}")
target_compile_options(argus_monitor PRIVATE "$<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fobjc-arc>")
target_link_libraries(argus_monitor PUBLIC nlohmann_json::nlohmann_json Threads::Threads
  "${ARGUS_FOUNDATION}" "${ARGUS_CORE_SERVICES}" PRIVATE argus_warnings)
add_library(argus_socket STATIC "${service}/SocketServer.cpp")
target_link_libraries(argus_socket PUBLIC Threads::Threads PRIVATE argus_warnings)
add_library(argus_ui STATIC "${service}/TerminalDisplayer.cpp")
target_link_libraries(argus_ui PUBLIC argus_monitor ftxui::component PRIVATE argus_warnings)

add_executable(ArgusMacServices "${service}/main.cpp")
target_link_libraries(ArgusMacServices PRIVATE argus_ui argus_socket argus_warnings)
add_executable(ArgusMacExtensionHelper "${helper}/main.cpp")
target_link_libraries(ArgusMacExtensionHelper PRIVATE nlohmann_json::nlohmann_json argus_warnings)
install(TARGETS ArgusMacServices ArgusMacExtensionHelper RUNTIME DESTINATION bin)
install(FILES "${ftxui_SOURCE_DIR}/LICENSE" DESTINATION share/licenses/argus/ftxui)
install(FILES "${nlohmann_json_SOURCE_DIR}/LICENSE.MIT" DESTINATION share/licenses/argus/nlohmann-json)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  function(argus_test_target name source)
    add_executable(${name} "native-tests/${source}")
    target_link_libraries(${name} PRIVATE argus_warnings)
    # Test assertions must also execute in a Release/Xcode multi-config build.
    target_compile_options(${name} PRIVATE -UNDEBUG)
  endfunction()
  argus_test_target(argus_transport_test transport.cpp)
  argus_test_target(argus_cpu_test cpu-usage.cpp)
  argus_test_target(argus_tabs_test tab-snapshots.cpp)
  target_link_libraries(argus_tabs_test PRIVATE nlohmann_json::nlohmann_json)
  argus_test_target(argus_concurrency_test concurrency.cpp)
  target_link_libraries(argus_concurrency_test PRIVATE argus_monitor)
  argus_test_target(argus_server_fixture server-fixture.cpp)
  target_link_libraries(argus_server_fixture PRIVATE argus_monitor argus_socket)
  argus_test_target(argus_browser_fixture browser-fixture.cpp)
  target_link_libraries(argus_browser_fixture PRIVATE nlohmann_json::nlohmann_json)
  add_test(NAME NativeTransport COMMAND argus_transport_test)
  add_test(NAME CpuUsage COMMAND argus_cpu_test)
  add_test(NAME TabSnapshots COMMAND argus_tabs_test)
  add_test(NAME ConcurrentMonitor COMMAND argus_concurrency_test)
  add_test(NAME HelperOffline COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native-tests/helper-offline.py" "$<TARGET_FILE:ArgusMacExtensionHelper>")
  add_test(NAME NativeIntegration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native-tests/integration.py" "$<TARGET_FILE:argus_server_fixture>" "$<TARGET_FILE:ArgusMacExtensionHelper>")
  add_test(NAME BrowserInstances COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native-tests/browser-instances.py" "$<TARGET_FILE:argus_server_fixture>" "$<TARGET_FILE:ArgusMacExtensionHelper>" "$<TARGET_FILE:argus_browser_fixture>")
  add_test(NAME NativeUI COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native-tests/ui-smoke.py" "$<TARGET_FILE:ArgusMacServices>" "$<TARGET_FILE:ArgusMacExtensionHelper>")
  set_tests_properties(NativeTransport CpuUsage TabSnapshots ConcurrentMonitor HelperOffline NativeIntegration BrowserInstances NativeUI PROPERTIES TIMEOUT 90)
endif()
